// Jellyfish Aquarium — BrainImation Visual
// Tentacle sway slows with high delta (deep rest / sleep)
// Added: Delta progress tracker + target line (for sleep training)
// Added: Gentle bell music driven by delta/alpha values


let jellyfish = [];
let deltaHistory = [];
let maxHistory = 200;
let targetDelta = 0.8; // Target delta level for sleep (0–1)


// Music variables
let synth;
let lastNoteTime = 0;


function setup() {
  colorMode(HSB, 360, 100, 100);
  noStroke();


  // Create multiple jellyfish
  for (let i = 0; i < 8; i++) {
    jellyfish.push({
      x: random(width),
      y: random(height * 0.2, height * 0.8),
      size: random(50, 100),
      swayOffset: random(TWO_PI),
      tentacleCount: floor(random(5, 10))
    });
  }


  // Setup gentle bell synth
  synth = new p5.PolySynth();
}


function draw() {
  // Real-time brain data
  let delta = constrain(eegData.delta, 0, 1);
  let alpha = constrain(eegData.alpha, 0, 1);
  let attention = constrain(eegData.attention, 0, 1);


  // Background — darker with higher delta
  let bgHue = map(delta, 0, 1, 200, 240);
  let bgBright = map(delta, 0, 1, 30, 10);
  background(bgHue, 30, bgBright);


  // Draw jellyfish
  for (let j of jellyfish) {
    drawJellyfish(j, delta, alpha, attention);
  }


  // Show delta level
  fill(0, 0, 100);
  textAlign(CENTER, CENTER);
  textSize(20);
  text("Delta: " + (delta * 100).toFixed(1) + "%", width / 2, 30);


  // Track and visualize delta progress
  updateDeltaHistory(delta);
  drawProgressGraph();


  // Play gentle music based on delta/alpha
  playGentleMusic(delta, alpha);
}


function drawJellyfish(j, delta, alpha, attention) {
  push();
  translate(j.x, j.y);


  let glow = map(alpha, 0, 1, 30, 80);
  let size = j.size * map(attention, 0, 1, 0.8, 1.3);


  // Body
  fill(200, 60, glow, 0.7);
  ellipse(0, 0, size, size * 0.7);


  // Tentacle sway speed slows with high delta
  let swaySpeed = map(delta, 0, 1, 0.2, 0.02);
  let time = frameCount * swaySpeed + j.swayOffset;


  stroke(200, 40, glow + 20);
  strokeWeight(1.5);
  noFill();


  // Tentacles
  for (let i = 0; i < j.tentacleCount; i++) {
    let xOffset = map(i, 0, j.tentacleCount - 1, -size / 3, size / 3);
    beginShape();
    for (let y = 0; y < size * 1.2; y += 5) {
      let wave = sin(time + y * 0.05 + i) * 5 * (1 - delta);
      curveVertex(xOffset + wave, y);
    }
    endShape();
  }


  pop();
}


// Track delta values over time
function updateDeltaHistory(delta) {
  deltaHistory.push(delta);
  if (deltaHistory.length > maxHistory) {
    deltaHistory.shift();
  }
}


// Draw delta progress graph and target line
function drawProgressGraph() {
  push();
  translate(width - 220, height - 120); // bottom-right corner


  // Background box
  noStroke();
  fill(0, 0, 0, 0.3);
  rect(0, 0, 200, 100, 5);


  // Label
  fill(0, 0, 100);
  textSize(12);
  textAlign(LEFT, TOP);
  text("Δ trend (sleep goal)", 8, 6);


  // Target delta line (sleep goal)
  let targetY = map(targetDelta, 0, 1, 90, 20);
  stroke(50, 30, 100);
  strokeWeight(1);
  for (let x = 0; x < 200; x += 8) {
    line(x, targetY, x + 4, targetY);
  }


  // Line graph for live delta
  stroke(200, 80, 100);
  noFill();
  beginShape();
  for (let i = 0; i < deltaHistory.length; i++) {
    let x = map(i, 0, maxHistory, 0, 200);
    let y = map(deltaHistory[i], 0, 1, 90, 20);
    vertex(x, y);
  }
  endShape();


  // Fill area under curve
  noStroke();
  fill(200, 60, 80, 0.3);
  beginShape();
  vertex(0, 100);
  for (let i = 0; i < deltaHistory.length; i++) {
    let x = map(i, 0, maxHistory, 0, 200);
    let y = map(deltaHistory[i], 0, 1, 90, 20);
    vertex(x, y);
  }
  vertex(200, 100);
  endShape(CLOSE);


  // Label the target line as "Sleep"
  fill(50, 30, 100);
  noStroke();
  textSize(10);
  textAlign(RIGHT, CENTER);
  text("Sleep", 190, targetY - 5);


  pop();
}


// --- ambient bell sounds ---
function playGentleMusic(delta, alpha) {
  if (!synth) return;


  // Only trigger a new note every ~1.2 seconds to keep it calm
  if (millis() - lastNoteTime < 1200) return;


  // Map delta (0–0.3) to soft bell note pitches (C4–G4)
  let notes = ['C4', 'D4', 'E4', 'G4'];
  let index = floor(map(delta, 0, 0.3, 0, notes.length));
  index = constrain(index, 0, notes.length - 1);
  let note = notes[index];


  // Volume based on alpha (0–1) for gentle amplitude
  let vel = map(alpha, 0, 1, 0.05, 0.12);


  // Play a soft bell with slow attack/release
  synth.play(note, vel, 0, 2.0);


  lastNoteTime = millis();
}

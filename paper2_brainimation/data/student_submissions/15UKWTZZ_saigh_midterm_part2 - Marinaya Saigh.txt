// === N170 ERP Averaging: Faces vs Objects (Fully Fixed) ===
// Alternates FACE / OBJECT, averages 4 EEG channels

const CHANNELS = ['TP9', 'AF7', 'AF8', 'TP10'];
const COLORS = {
  face: color(255, 200, 200),    // pink
  object: color(128, 0, 128)   // purple
};

const PRE_MS = 200;
const POST_MS = 800;
const SR = 256;
const EPOCH_LEN = Math.round((PRE_MS + POST_MS) / 1000 * SR);
const MAX_TRIALS = 20;

let epochs = { face: {}, object: {} };
let trialCount = { face: 0, object: 0 };
for (let cond in epochs) {
  for (let ch of CHANNELS) epochs[cond][ch] = [];
}

let showFace = true;
let trial = 0;
let showStim = true;
let lastSwitch = 0;
let state = "stimulus";
let stateStart = 0;

function setup() {
  colorMode(HSB, 360, 100, 100);
  frameRate(2); // same as the working example
  stateStart = millis();
  textAlign(CENTER, CENTER);
}

function draw() {
  background(0, 0, 8);
  drawERPPanel();
  drawStimulusPanel();
  updateState();
}

// ---------------- ERP PANEL ----------------
function drawERPPanel() {
  const h = height / 2;
  const spacing = 60;

  // title
  fill(0, 0, 100);
  textAlign(CENTER, TOP);
  textSize(16);
  text("ERP Average (Faces vs Objects)", width / 2, 10);
  textSize(12);
  text("Averaged across TP9, AF7, AF8, TP10", width / 2, 30);

  // plot each channel
  for (let i = 0; i < CHANNELS.length; i++) {
    const ch = CHANNELS[i];
    const baseY = 70 + i * spacing;

    stroke(50);
    line(60, baseY, width - 60, baseY);

    for (let cond of ['face', 'object']) {
      const avg = computeAverage(epochs[cond][ch]);
      if (avg.length === EPOCH_LEN) {
        stroke(COLORS[cond]);
        strokeWeight(2);
        noFill();
        beginShape();
        for (let j = 0; j < avg.length; j++) {
          const t = map(j, 0, EPOCH_LEN - 1, -PRE_MS, POST_MS);
          const x = map(t, -PRE_MS, POST_MS, 60, width - 60);
          const y = baseY - map(avg[j], -50, 50, -25, 25);
          vertex(x, y);
        }
        endShape();
      }
    }

    // label
    noStroke();
    fill(0, 0, 90);
    textAlign(LEFT, CENTER);
    textSize(12);
    text(ch, 15, baseY);
  }

  // time axis -200 to +800
  stroke(60);
  for (let t = -PRE_MS; t <= POST_MS; t += 100) {
    const x = map(t, -PRE_MS, POST_MS, 60, width - 60);
    stroke(40);
    line(x, 50, x, h - 20);
    noStroke();
    fill(120);
    textAlign(CENTER, BOTTOM);
    text(`${t} ms`, x, h - 25);
  }

  // 0 ms line
  stroke(0, 0, 100);
  const zx = map(0, -PRE_MS, POST_MS, 60, width - 60);
  line(zx, 50, zx, h - 20);

  // legend
  noStroke();
  textAlign(RIGHT, TOP);
  textSize(12);
  fill(COLORS.face);
  text(`face (n=${trialCount.face})`, width - 160, 10);
  fill(COLORS.object);
  text(`object (n=${trialCount.object})`, width - 60, 10);
}

// ---------------- STIMULUS PANEL ----------------
function drawStimulusPanel() {
  const h = height / 2;
  const cx = width / 2;
  const cy = h + h / 2;

  const cond = showFace ? "FACE" : "OBJECT";

  fill(0, 0, 90);
  textSize(16);
  textAlign(CENTER, TOP);
  text(cond, cx, h + 10);
  text(`Trial ${trial + 1}`, cx, h + 30);

  if (state === "stimulus") {
    if (showFace) drawFace(cx, cy);
    else drawObject(cx, cy);
  }
}

function drawFace(x, y) {
  fill(60, 80, 100);
  ellipse(x, y, 120, 150);
  fill(0);
  ellipse(x - 25, y - 20, 15, 20);
  ellipse(x + 25, y - 20, 15, 20);
  noFill();
  stroke(0);
  strokeWeight(3);
  arc(x, y + 30, 60, 40, 0, PI);
}

function drawObject(x, y) {
  noStroke();
  fill(0, 0, 100);
  rectMode(CENTER);
  rect(x, y, 120, 120, 10);
}

// ---------------- STATE MACHINE ----------------
function updateState() {
  const now = millis();

  // every draw call toggles between showing and ITI
  if (frameCount % 2 === 0) {
    // capture EEG epoch for current condition
    const cond = showFace ? "face" : "object";
    captureEEG(cond);
    trial++;
    showFace = !showFace;
  }
}

// ---------------- EEG CAPTURE ----------------
function captureEEG(cond) {
  let data;
  try {
    data = eegData.getAllChannels(EPOCH_LEN);
  } catch (e) {
    data = simulateEEG();
  }

  for (let ch of CHANNELS) {
    const raw = data[ch];
    const baseCount = Math.round(PRE_MS / 1000 * SR);
    const baseline = raw.slice(0, baseCount);
    const meanBase = baseline.reduce((a, b) => a + b, 0) / baseline.length;
    const epoch = raw.map(v => v - meanBase);

    epochs[cond][ch].push(epoch);
    if (epochs[cond][ch].length > MAX_TRIALS) epochs[cond][ch].shift();
  }

  trialCount[cond]++;
}

// ---------------- AVERAGING ----------------
function computeAverage(list) {
  if (!list || list.length === 0) return new Array(EPOCH_LEN).fill(0);
  const avg = new Array(EPOCH_LEN).fill(0);
  for (let i = 0; i < EPOCH_LEN; i++) {
    let sum = 0;
    for (let k = 0; k < list.length; k++) sum += list[k][i] || 0;
    avg[i] = sum / list.length;
  }
  return avg;
}

// ---------------- SIMULATED EEG ----------------
function simulateEEG() {
  let data = {};
  for (let ch of CHANNELS) {
    data[ch] = [];
    for (let i = 0; i < EPOCH_LEN; i++) {
      data[ch].push(
        10 * sin(TWO_PI * i / 40 + millis() / 1000 + random(-0.3, 0.3)) +
        random(-5, 5)
      );
    }
  }
  return data;
}
